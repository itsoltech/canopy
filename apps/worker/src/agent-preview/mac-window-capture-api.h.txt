#define SELECT(name) sel_registerName(name)
#define SEND(RETURN, ...) ((RETURN (*)(Object, Object, ##__VA_ARGS__))objc_msgSend)

static Object get(Object receiver, const char *selector) { return SEND(Object)(receiver, SELECT(selector)); }
static Object make(const char *cls) { return get(objc_getClass(cls), "new"); }
static int responds(Object receiver, const char *name) {
    return receiver && SEND(_Bool, Object)(receiver, SELECT("respondsToSelector:"), SELECT(name));
}
static long integer(Object number) { return SEND(long)(number, SELECT("integerValue")); }
static void setBool(Object obj, const char *selector, _Bool value) {
    SEND(void, _Bool)(obj, SELECT(selector), value);
}
static void setSize(Object obj, const char *selector, unsigned long value) {
    SEND(void, unsigned long)(obj, SELECT(selector), value);
}

static Rect rectProperty(Object obj, const char *selector) {
#ifdef __x86_64__
    // The Intel Objective-C ABI uses a hidden destination argument for a CGRect return value.
    extern void objc_msgSend_stret(void);
    Rect result;
    ((void (*)(Rect *, Object, Object))objc_msgSend_stret)(&result, obj, SELECT(selector));
    return result;
#else
    return SEND(Rect)(obj, SELECT(selector));
#endif
}

// The public Apple Blocks ABI for a global, capture-free block. ScreenCaptureKit invokes it
// on a dispatch queue. Only this isolated helper waits. Killing it cancels a pending pull.
static Object completionGate, completedValue;
static void complete(Object block, Object value, Object error) {
    (void)block;
    completedValue = value && !error ? CFRetain(value) : 0;
    dispatch_semaphore_signal(completionGate);
}
static struct {
    unsigned long reserved, size;
} descriptor;
static struct {
    Object isa;
    int flags, reserved;
    void (*invoke)(Object, Object, Object);
    Object descriptor;
} completion;

static int initializeCompletion(void) {
    completion.isa = dlsym((Object)-2, "_NSConcreteGlobalBlock"); // RTLD_DEFAULT
    completion.flags = 1 << 28;                                   // BLOCK_IS_GLOBAL
    completion.invoke = complete;
    descriptor.size = sizeof(completion);
    completion.descriptor = &descriptor;
    completionGate = dispatch_semaphore_create(0);
    return completion.isa && completionGate;
}
static Object waitForCompletion(void) {
    // Also bound an outstanding system call if the worker itself exits while awaiting a frame.
    if (dispatch_semaphore_wait(completionGate, dispatch_time(0, 2500000000LL)))
        _exit(1);
    Object value = completedValue;
    completedValue = 0;
    return value;
}
